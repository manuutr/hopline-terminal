fgt-edge1 # show system interface port2
config system interface
    edit "port2"
        set vdom "root"
        set ip 10.0.2.1 255.255.255.252
        set allowaccess ping
        set type physical
        set alias "to-core-sw1"
        set lldp-reception enable
        set snmp-index 2
    next
end
